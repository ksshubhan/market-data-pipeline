5. Minimal reproducer: one blocking cv.wait, no project code.
#include <condition_variable>
#include <mutex>
#include <thread>
int main()
{
    std::mutex m;
    std::condition_variable cv;
    bool ready = false;
    std::thread t([&] {
        std::unique_lock<std::mutex> lock(m);
        cv.wait(lock, [&] { return ready; });
    });
    {
        std::lock_guard<std::mutex> lock(m);
        ready = true;
    }
    cv.notify_one();
    t.join();
    return 0;
}
